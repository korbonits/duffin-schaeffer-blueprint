\chapter{Overview}
\label{chap:overview}

Let $\ps : \N \to \R_{\geq 0}$. The Duffin--Schaeffer conjecture, posed in 1941 and
proved by Koukoulopoulos and Maynard in 2019, says that if
\[ \sum_{q \geq 1} \frac{\ps(q)\varphi(q)}{q} = \infty \]
then almost every $\alpha \in [0,1]$ admits infinitely many coprime solutions to
\[ \left| \alpha - \frac{a}{q} \right| \leq \frac{\ps(q)}{q}. \tag{1.7} \]

This blueprint charts a formalization of that theorem, together with the
non-coprime dichotomy of the paper's Theorem 2. The terminal nodes are transcribed
verbatim from \texttt{ImperialCollegeLondon/AnnalsChallenge}, where they stand as
\texttt{sorry}; discharging them here discharges them there.

\section*{Why this target}

Mathlib has already done the first reduction and has left the gap explicitly marked.
\texttt{Mathlib/NumberTheory/WellApproximable.lean} (Oliver Nash, 2022) proves
\emph{Gallagher's ergodic theorem}: for any sequence of distances, the
well-approximable set is null or co-null. Its own module docstring reads:

\begin{quote}
Given a particular $\delta$, the Duffin--Schaeffer conjecture (now a theorem) gives a
criterion for deciding which of the two cases in the conclusion of Gallagher's theorem
actually occurs. \dots\ We do \emph{not} include a formalisation of the
Koukoulopoulos--Maynard result here.
\end{quote}

That is the hole this project fills. Because Gallagher gives a zero--one law, it
suffices to prove that the set has \emph{positive} measure --- and therefore no
constant anywhere in this blueprint ever has to be computed. That single observation
governs the shape of every estimate below.

\section*{Scope}

Theorem 1 and Theorem 2(a),(b) are in scope. Corollary 3 --- the Hausdorff dimension
--- is not, and Chapter \ref{chap:hausdorff} says why: it needs mass transference,
which is an independent project of comparable size that would roughly double the work
while adding nothing to the conjecture itself.

\section*{How to read the status marks}

A node marked \notready\ has no Lean statement yet. Some of those are merely unwritten;
others are marked \textbf{TRANSCRIBE}, which means something stronger --- the statement
involves explicit exponents, normalisations or thresholds that must be copied from the
source rather than reconstructed. A plausible-looking Lean statement with a wrong
constant is worse than a visible gap, so those nodes carry their section reference and
nothing else until someone has the paper open.

\chapter{The approximation sets}
\label{chap:setup}

\begin{definition}[Inequality (1.7)]
  \label{def:approximates}
  \lean{DuffinSchaeffer.Approximates}
  \leanok
  For $\ps : \N_{>0} \to \R_{\geq 0}$, $\alpha \in \R$, $a \in \N$ and
  $q \in \N_{>0}$, write $\mathrm{Approximates}(\ps, \alpha, a, q)$ for
  $\left| \alpha - a/q \right| \leq \ps(q)/q$.
\end{definition}

\begin{definition}[The sets $\Aq$]
  \label{def:setaq}
  \lean{DuffinSchaeffer.setAq}
  \leanok
  \uses{def:approximates}
  $\Aq = \{ \alpha \in [0,1] : \exists a,\ \gcd(a,q) = 1 \text{ and } (1.7) \text{ holds} \}$.
\end{definition}

\begin{definition}[$\A(\ps)$ and $\K(\ps)$]
  \label{def:seta}
  \lean{DuffinSchaeffer.setA, DuffinSchaeffer.setK}
  \leanok
  \uses{def:approximates}
  $\A(\ps)$ is the set of $\alpha \in [0,1]$ for which (1.7) has infinitely many
  coprime solutions $(a,q)$; $\K(\ps)$ is the same with coprimality dropped and
  $0 \leq a \leq q$ imposed.
\end{definition}

\begin{definition}[The radius $\Psi$ and the majorant $\pstar$]
  \label{def:psistar}
  \lean{DuffinSchaeffer.psiSup, DuffinSchaeffer.psiStar}
  \leanok
  $\Psi(q) = \sup \{ \ps(n)/n : q \mid n \}$ and $\pstar(q) = \varphi(q)\Psi(q)$.
  The supremum is taken in $[0,\infty]$, so it always exists.
\end{definition}

\begin{lemma}[$\Psi \leq \pstar$]
  \label{lem:psisup-le-psistar}
  \lean{DuffinSchaeffer.psiSup_le_psiStar}
  \leanok
  \uses{def:psistar}
  $\Psi(q) \leq \pstar(q)$, and hence $\sum_q \Psi(q) < \infty$ whenever
  $\sum_q \pstar(q) < \infty$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} $\varphi(q) \geq 1$. Small, but it is what makes the covering argument
  of Theorem \ref{thm:two-a} work: the convergence hypothesis has to control the
  \emph{radii} $\Psi(d)$ and not only the weighted radii $\varphi(d)\Psi(d)$.
  Lean: \texttt{tsum\_psiSup\_lt\_top}.
\end{proof}

\begin{lemma}[$\Aq$ as a union of balls]
  \label{lem:setaq-union}
  \lean{DuffinSchaeffer.setAq_eq}
  \leanok
  \uses{def:setaq}
  $\Aq = [0,1] \cap \bigcup_{\gcd(a,q)=1} \overline{B}(a/q, \ps(q)/q)$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} Unfolding both sides; (1.7) is membership of a closed ball.
\end{proof}

\begin{lemma}[Measurability of $\Aq$]
  \label{lem:measurable-setaq}
  \lean{DuffinSchaeffer.measurableSet_setAq}
  \leanok
  \uses{lem:setaq-union}
  $\Aq$ is measurable.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} A countable union of closed balls, intersected with $[0,1]$.
  No hypothesis on $\ps$ is needed.
\end{proof}

\begin{lemma}[Measure of a single $\Aq$]
  \label{lem:volume-setaq}
  \lean{DuffinSchaeffer.volume_setAq_le}
  \leanok
  \uses{lem:setaq-union}
  If $\ps(q) \leq 1/2$ then $\Leb(\Aq) \leq 2\ps(q)\varphi(q)/q$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved}, for the upper bound. Under $\ps(q) \leq 1/2$ no numerator outside
  $0 \leq a \leq q$ can bring a point of $[0,1]$ within $\ps(q)/q$ of $a/q$
  (\texttt{setAq\_subset}), so $\Aq$ is covered by $\#\{a \le q : \gcd(a,q)=1\}$ balls of
  radius $\ps(q)/q$; for $q \geq 2$ that count is exactly $\varphi(q)$, because $a = 0$
  and $a = q$ are both excluded by coprimality (\texttt{card\_numerators}).

  \emph{The hypothesis is not cosmetic, and this lemma was originally stated without it.}
  Two things go wrong. For large $\ps(q)$ the numerators $a > q$ also deposit mass in
  $[0,1]$, so the count $\varphi(q)$ is not an upper bound. And at $q = 1$ both $a = 0$
  and $a = 1$ are coprime to $1$ while $\varphi(1) = 1$, so $\#\{a \le q\} = \varphi(q)$
  simply fails; the conclusion still holds there, but only because the two intervals are
  truncated by $[0,1]$ to total length exactly $2\ps(1)$, which is a different argument
  (\texttt{volume\_setAq\_le\_one}). Both problems vanish under $\ps \leq 1/2$, which
  Lemma \ref{lem:truncate} supplies and everything downstream assumes anyway.

  The matching lower bound is now also proved, as
  \texttt{volume\_setAq\_of\_two\_le}: for $q \geq 2$ every ball lies inside $[0,1]$, so
  the containment above is an equality, and the balls are pairwise \emph{almost}
  disjoint --- at $\ps(q) = 1/2$ consecutive intervals share an endpoint, so the
  statement needed is a null intersection (\texttt{volume\_closedBall\_inter}) rather
  than disjointness, and \texttt{measure\_biUnion\_finset}$_0$ then gives
  $\Leb(\Aq) = 2\ps(q)\varphi(q)/q$ exactly.
\end{proof}

\begin{lemma}[The numerator bound]
  \label{lem:le-of-approximates}
  \lean{DuffinSchaeffer.le_of_approximates}
  \leanok
  \uses{def:approximates}
  If $\ps(q) \leq 1/2$ and $\alpha \in [0,1]$ satisfies (1.7), then $a \leq q$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} $a/q \leq \alpha + \ps(q)/q \leq 1 + 1/(2q)$, so $a \leq q + 1/2$.
  This is the single place the truncation hypothesis does real work; both the covering
  bound and the pair-versus-denominator count rest on it.
\end{proof}

\begin{lemma}[Monotonicity in $\ps$]
  \label{lem:seta-mono}
  \lean{DuffinSchaeffer.setA_mono}
  \leanok
  \uses{def:seta}
  If $\ps \leq \ps'$ pointwise then $\A(\ps) \subseteq \A(\ps')$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} Enlarging the tolerance can only add solutions.
\end{proof}

\begin{lemma}[Reduction to $\ps \leq 1/2$]
  \label{lem:truncate}
  \lean{DuffinSchaeffer.setA_truncate}
  \leanok
  \uses{lem:seta-mono, thm:gallagher}
  Assume $\Leb(\A(\ps)) \in \{0,1\}$. Then either $\Leb(\A(\ps)) = 1$, or replacing
  $\ps$ by $\min(\ps, 1/2)$ leaves the measure unchanged.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved}, but note what it is proved from.

  \emph{This is not a Chapter 2 result, and presenting it as one was a mistake in the
  first draft of this blueprint.} Either disjunct needs to know that $\A(\ps)$ cannot
  have intermediate measure --- which is Gallagher's zero-one law, a Chapter
  \ref{chap:gallagher} input. So the zero-one law appears here as an explicit
  hypothesis rather than being smuggled in. There is no circularity: every statement in
  Chapter \ref{chap:gallagher} carries the hypothesis $\ps \leq 1/2$ and none of them
  uses this lemma.

  The tempting elementary route fails. One would like to argue that if $\ps(q) > 1/2$
  for infinitely many $q$ then $\A(\ps)$ is full outright, but that is false: at $q = 6$
  only $b = 1, 5$ are coprime to $6$, so the balls of radius $\ps(6)/6$ around $1/6$ and
  $5/6$ cover $[0,1]$ only once $\ps(6) \geq 2$. Given the zero-one law the proof is two
  lines --- if the measure is $0$ then monotonicity (Lemma \ref{lem:seta-mono}) forces
  the truncation to be null as well --- but nothing weaker closes the gap.
\end{proof}

\begin{lemma}[$\A$ as a $\limsup$ of the $\Aq$]
  \label{lem:seta-limsup}
  \lean{DuffinSchaeffer.mem_setA_iff}
  \leanok
  \uses{def:seta, def:setaq, lem:le-of-approximates}
  For $\ps \leq 1/2$, $\alpha \in \A(\ps)$ iff $\alpha \in [0,1]$ and $\alpha$ lies in
  infinitely many $\Aq$.
\end{lemma}

\begin{proof}
  The content is that infinitely many \emph{pairs} $(a,q)$ forces infinitely many
  \emph{denominators}. Under $\ps \leq 1/2$ a solution has $0 \leq a \leq q$, so each
  denominator admits only finitely many numerators. Without that hypothesis the
  implication genuinely fails, which is why Lemma \ref{lem:truncate} comes first.
  Lean: \texttt{mem\_setA\_iff}.
\end{proof}

\begin{lemma}[Measurability of $\A$ and $\K$]
  \label{lem:measurable-seta}
  \lean{DuffinSchaeffer.measurableSet_setA, DuffinSchaeffer.measurableSet_setK}
  \leanok
  \uses{def:seta}
  $\A(\ps)$ and $\K(\ps)$ are measurable.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} Both are $[0,1]$ intersected with a set of the form
  $\{\alpha : \{i : \alpha \in B_i\} \text{ infinite}\}$ for a countable family --- here
  indexed by pairs $(a,q)$, each $B_{(a,q)}$ a closed ball or empty. Such a set equals
  $\bigcap_{t} \bigcup_{i \notin t} B_i$ over the finite subsets $t$, which is a countable
  intersection of countable unions because $\mathrm{Finset}\,\iota$ is countable when
  $\iota$ is. Stated once as \texttt{measurableSet\_setOf\_infinite} and used for both.
  No hypothesis on $\ps$ is needed.
\end{proof}

\begin{lemma}[Divergence transfers to the measures]
  \label{lem:divergence}
  \lean{DuffinSchaeffer.not_summable_volume_setAq}
  \leanok
  \uses{lem:volume-setaq}
  If $\sum \ps(q)\varphi(q)/q$ diverges and $\ps \leq 1/2$, then
  $\sum_q \Leb(\Aq) = \infty$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} Immediate from the equality in Lemma \ref{lem:volume-setaq} once the
  $q = 1$ term is dealt with. It is absorbed rather than computed: the equality is
  available for $q \geq 2$, and a single finite term cannot rescue a divergent series,
  so bounding $\Leb(\Aq)$ below by the $q \ge 2$ formula plus an indicator at $q = 1$
  suffices.
\end{proof}

\begin{theorem}[Gallagher's ergodic theorem]
  \label{thm:gallagher}
  \lean{AddCircle.addWellApproximable_ae_empty_or_univ}
  \leanok
  \mathlibok
  For any $\delta : \N \to \R$ tending to $0$, the set
  $\mathrm{addWellApproximable}(\R/\Z, \delta)$ is either null or co-null.
\end{theorem}

\begin{proof}
  \leanok
  In Mathlib, \texttt{Mathlib/NumberTheory/WellApproximable.lean}. Nothing to do.
\end{proof}

\begin{definition}[The distance sequence]
  \label{def:distseq}
  \lean{DuffinSchaeffer.distSeq}
  \leanok
  $\delta_\ps(n) = \ps(n)/n$ for $n > 0$, and $0$ at $n = 0$.
\end{definition}

\begin{lemma}[Dropping the vanishing hypothesis]
  \label{lem:gallagher-no-tendsto}
  \notready
  \uses{thm:gallagher}
  \emph{Not needed on the critical path} --- see Proposition
  \ref{prop:positive-suffices}. It is required only by Lemma \ref{lem:truncate} at
  general $\ps$.
  Gallagher's theorem holds for any non-negative $\delta$, with no assumption that
  $\delta \to 0$.
\end{lemma}

\begin{proof}
  Mathlib's own TODO records the argument: if $\delta \not\to 0$ then
  $\mathrm{addWellApproximable}(\R/\Z,\delta)$ is all of $\R/\Z$, by an elementary
  non-measure-theoretic argument. The Duffin--Schaeffer hypothesis does not supply
  $\delta \to 0$, so this cannot be skipped. It is a self-contained Mathlib
  contribution. Lean: \texttt{addWellApproximable\_ae\_empty\_or\_univ\_of\_nonneg}.
\end{proof}

\begin{lemma}[The strict form]
  \label{lem:strict}
  \lean{DuffinSchaeffer.setA_ae_eq_setAStrict}
  \leanok
  \uses{def:seta}
  $\A(\ps)$ agrees almost everywhere with its strict-inequality analogue, in which (1.7)
  is required with $<$ rather than $\leq$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} Mathlib's \texttt{addWellApproximable} uses a strict inequality while
  (1.7) uses $\leq$, so the two have to be reconciled before any transfer can happen.

  If $\alpha$ satisfies infinitely many instances of (1.7) with $\leq$ but only finitely
  many with $<$, then infinitely many instances hold with \emph{equality}, so $\alpha$ is
  one of the countably many endpoints $a/q \pm \ps(q)/q$. Hence the difference is
  contained in a countable set and is null. This is the one step of the transfer where an
  error would not be visible; the rest is bookkeeping about the quotient map.
\end{proof}

\begin{lemma}[The circle norm against the absolute value]
  \label{lem:circle-norm}
  \lean{DuffinSchaeffer.norm_lt_iff_abs_lt}
  \leanok
  For $n \geq 2$, $1 \leq m \leq n-1$, $\alpha \in [0,1]$ and $\delta \leq 1/(2n)$,
  \[ \|\alpha - m/n\|_{\R/\Z} < \delta \iff |\alpha - m/n| < \delta. \]
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} The two do not agree in general --- $\alpha = 0.9$ and $m/n = 0.1$ are
  at circle distance $0.2$ and real distance $0.8$ --- so the content is that under
  $\ps \leq 1/2$ the discrepant case never produces a solution.

  Mathlib's \texttt{AddCircle.norm\_coe\_eq\_abs\_iff} gives agreement whenever
  $|\alpha - m/n| \leq 1/2$. Otherwise, counting bounds the real distance:
  $m/n$ lies in $[1/n, 1 - 1/n]$ and $\alpha \in [0,1]$, so
  $|\alpha - m/n| \leq 1 - 1/n$, and then the circle distance is
  $1 - |\alpha - m/n| \geq 1/n > \delta$. Neither side holds, so the equivalence is
  vacuously true there.

  This is where the truncation $\ps \leq 1/2$ pays for itself a second time; the first
  was Lemma \ref{lem:le-of-approximates}.
\end{proof}

\begin{lemma}[Transfer to the circle]
  \label{lem:transfer}
  \lean{DuffinSchaeffer.setA_eq_preimage_addWellApproximable}
  \leanok
  \uses{def:seta, def:distseq, lem:strict, lem:circle-norm}
  For $\ps \leq 1/2$, $\A(\ps)$ agrees up to a null set with the preimage of
  $\mathrm{addWellApproximable}(\R/\Z, \delta_\ps)$ under $\R \to \R/\Z$,
  intersected with $[0,1]$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} In fact the \emph{strict} form is exactly the preimage --- no null
  sets at all (\texttt{setAStrict\_eq\_preimage}) --- and Lemma \ref{lem:strict} then
  supplies the only null set in sight.

  Three mismatches had to be reconciled, denominator by denominator
  (\texttt{mem\_setAqStrict\_iff\_circle}). The numerator range: Mathlib asks for
  $m < n$ while (1.7) bounds nothing, but coprimality plus $\ps \le 1/2$ confine $a$ to
  $[1, q-1]$ for $q \geq 2$. The norm: Lemma \ref{lem:circle-norm}. And $q = 1$, where
  Mathlib's only admissible numerator is $m = 0$ while the real condition also allows
  $a = 1$ --- they agree because $0/1$ and $1/1$ are the same point of the circle. The
  index sets then match by $\N_{>0} \hookrightarrow \N$, whose image misses only $n = 0$,
  where the condition $\exists m < 0$ is vacuous.
\end{proof}

\begin{proposition}[Positive measure suffices]
  \label{prop:positive-suffices}
  \lean{DuffinSchaeffer.volume_setA_eq_one_of_pos}
  \leanok
  \uses{thm:gallagher, lem:transfer}
  If $\Leb(\A(\ps)) > 0$ then $\Leb(\A(\ps)) = 1$.
\end{proposition}

\begin{proof}
  \leanok
  \textbf{Proved.} Under $\ps \leq 1/2$ the distances satisfy $\delta_\ps(n) \leq 1/(2n)$
  and so tend to $0$, which means Mathlib's Gallagher applies \emph{as stated}: Lemma
  \ref{lem:gallagher-no-tendsto} is not needed on this path at all. Given the dichotomy,
  transport it across Lemma \ref{lem:transfer} using
  \texttt{AddCircle.add\_projection\_respects\_measure}, which computes the circle measure
  of a set as the Lebesgue measure of its preimage in $(0,1]$; the difference between
  $[0,1]$ and $(0,1]$ is a single point.

  This is the proposition that makes every constant in the rest of the blueprint
  irrelevant.
\end{proof}

\chapter{The second moment method}
\label{chap:secondmoment}

\begin{lemma}[Cauchy--Schwarz for \texttt{lintegral}]
  \label{lem:cauchy-schwarz}
  \lean{DuffinSchaeffer.lintegral_mul_sq_le}
  \leanok
  $\left(\int^- fg\right)^2 \leq \int^- f^2 \cdot \int^- g^2$ for
  $\mathbb{R}_{\geq 0}^{\infty}$-valued $f, g$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} Mathlib has no Cauchy--Schwarz inequality for measure-theoretic
  integrals --- searching \texttt{MeasureTheory} and \texttt{Probability} for
  \texttt{cauchy\_schwarz} returns nothing. The nearest available statement is
  \texttt{ENNReal.lintegral\_mul\_le\_Lp\_mul\_Lq}, H\"older with real exponents, whose
  $p = q = 2$ case is Cauchy--Schwarz but phrased through \texttt{ENNReal.rpow}. The work
  is converting that to the squared form an application actually wants. Independently
  upstreamable.
\end{proof}

\begin{lemma}[Chung--Erd\H{o}s]
  \label{lem:chung-erdos}
  \lean{DuffinSchaeffer.chung_erdos}
  \leanok
  \uses{lem:cauchy-schwarz}
  For finitely many measurable events,
  $\left(\sum_i \mu(A_i)\right)^2 \leq \mu\!\left(\bigcup_i A_i\right) \sum_{i,j} \mu(A_i \cap A_j)$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} Let $F = \sum_i \mathbf{1}_{A_i}$ be the counting function and
  $U = \bigcup_i A_i$. Then $\int^- F = \sum_i \mu(A_i)$, and expanding the square of the
  finite sum gives $\int^- F^2 = \sum_{i,j} \mu(A_i \cap A_j)$. Since $F$ vanishes off
  $U$ we have $F = \mathbf{1}_U F$ pointwise, and Lemma \ref{lem:cauchy-schwarz} applied
  to that pair is the claim, using $\mathbf{1}_U^2 = \mathbf{1}_U$.

  No finiteness hypothesis on $\mu$ is needed, so the statement was weakened accordingly.
\end{proof}

\begin{proposition}[Second moment, escaping index sets]
  \label{prop:quasi-bc-escaping}
  \lean{DuffinSchaeffer.measure_infinite_pos_of_escaping}
  \leanok
  \uses{lem:chung-erdos}
  Let $(A_i)_{i \in \N_{>0}}$ be measurable and suppose there are a constant $C$ and
  finite sets $S_n$ with pair correlations bounded by $C$ times the square of the first
  moment, and such that for every $k$ some $S_n$ consists entirely of indices exceeding
  $k$ and carries positive mass. Then the set of points lying in infinitely many $A_i$
  has measure at least $C^{-1}$.
\end{proposition}

\begin{proof}
  \leanok
  \textbf{Proved.} Chung--Erd\H{o}s on $S_n$ gives
  $a^2 \leq \mu(\bigcup_{i \in S_n} A_i)\,C\,a^2$ where $a$ is the first moment; $a$ is
  nonzero by hypothesis and finite because $\mu$ is a probability measure and the sum is
  finite, so cancelling gives $1 \leq \mu(\bigcup_{i \in S_n} A_i)\,C$. The escape
  hypothesis places that union inside the tail $\bigcup_{i > k} A_i$, and continuity from
  above along the tails --- whose intersection is exactly the set of points lying in
  infinitely many $A_i$ --- finishes.

  This is where the constant $C$ finally becomes a lower bound on a measure, and where
  the remark of Chapter \ref{chap:overview} is discharged: any positive bound will do,
  because Proposition \ref{prop:positive-suffices} then upgrades it to $1$.
\end{proof}

\begin{proposition}[Divergence Borel--Cantelli, quasi-independent form]
  \label{prop:quasi-bc}
  \lean{DuffinSchaeffer.measure_infinite_pos_of_overlap}
  \leanok
  \uses{lem:chung-erdos}
  As in Proposition \ref{prop:quasi-bc-escaping}, but assuming only that the first moments
  $\sum_{i \in S_n} \mu(A_i)$ tend to infinity, with no escape hypothesis. The conclusion
  is that the set of points lying in infinitely many $A_i$ has \emph{positive} measure.
\end{proposition}

\begin{proof}
  \leanok
  \textbf{Proved.} Note the conclusion: Proposition \ref{prop:quasi-bc-escaping} gives the
  sharp constant $C^{-1}$, while here the bound degrades to $(4C)^{-1}$, so the statement
  is phrased as positivity. That is all any consumer needs --- Proposition
  \ref{prop:positive-suffices} turns any positive lower bound into $1$, and the whole
  development is arranged so that no constant is ever computed.

  The weakening is what makes the proof elementary. Fix $k$, let $t$ be the initial
  segment up to $k$, and put $K = \sum_{i \in t} \mu(A_i)$, a fixed finite quantity. The
  sharp version wants $x^2 \leq MC(x+K)^2$ with $x = \sum_{i \in S_n \setminus t} \mu(A_i)
  \to \infty$, hence a ratio and a limit, both awkward in
  $\mathbb{R}_{\geq 0}^{\infty}$. Settling for a factor of $4$ removes them: choose $n$
  with $\sum_{i \in S_n} \mu(A_i) \geq 2K + 1$, which forces $K \leq x$ and hence
  $\sum_{i \in S_n} \mu(A_i) \leq 2x$; Chung--Erd\H{o}s on $S_n \setminus t$ then gives
  $x^2 \leq 4 M C x^2$ directly, and $x$ cancels.
\end{proof}

\chapter{Anatomy of integers}
\label{chap:anatomy}

Nothing in this chapter is due to Koukoulopoulos and Maynard; it is the standard
toolkit their argument assumes, and Mathlib does not have it. Every node is
independently upstreamable and none depends on the rest of the project.

\begin{lemma}[First reduction toward Mertens]
  \label{lem:mertens-reduction}
  \lean{DuffinSchaeffer.abs_sum_log_sub_le}
  \leanok
  \uses{lem:dirichlet-swap}
  $\left| \sum_{q \leq n} \log q - n \sum_{d \leq n} \Lambda(d)/d \right|
   \leq (\log 4 + 4)\, n$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} This is the step the Dirichlet swap (Lemma \ref{lem:dirichlet-swap})
  was built for. Summing von Mangoldt over divisors gives $\log$
  (Mathlib's \texttt{vonMangoldt\_sum}), so the swap turns $\sum_{q \leq n} \log q$ into
  $\sum_{d \leq n} \lfloor n/d \rfloor \Lambda(d)$. Replacing $\lfloor n/d \rfloor$ by
  $n/d$ costs at most $\sum_{d \le n} \Lambda(d) = \psi(n)$, and Mathlib's Chebyshev bound
  \texttt{psi\_le\_const\_mul\_self} gives $\psi(n) \le (\log 4 + 4)n$.
\end{proof}

\begin{lemma}[Mertens' first theorem]
  \label{lem:mertens-one}
  \notready
  \uses{lem:mertens-reduction}
  $\sum_{d \leq n} \Lambda(d)/d = \log n + O(1)$.
\end{lemma}

\begin{proof}
  What is missing is the other side of the identity of Lemma
  \ref{lem:mertens-reduction}, namely $\sum_{q \leq n} \log q = n\log n + O(n)$ --- that
  is, Stirling. Mathlib has \texttt{Stirling.stirlingSeq} and its convergence, so this is
  assembly rather than invention, but it is not done.
\end{proof}

\begin{lemma}[Mertens' second theorem]
  \label{lem:mertens-two}
  \notready
  \uses{lem:mertens-one}
  $\sum_{p < n} 1/p = \log\log n + M + O(1/\log n)$.

  From Lemma \ref{lem:mertens-one} by Abel summation, for which Mathlib has
  \texttt{NumberTheory/AbelSummation}. Lean: \texttt{mertens\_second}.
\end{lemma}

\begin{lemma}[Mertens' third theorem]
  \label{lem:mertens-three}
  \notready
  \uses{lem:mertens-two}
  $\prod_{p < n} (1 - 1/p)^{-1} \asymp \log n$. The sharp constant $e^{\gamma}$ is
  never needed. Lean: \texttt{mertens\_third}.
\end{lemma}

\begin{lemma}[Divisor bound]
  \label{lem:divisor-bound}
  \lean{DuffinSchaeffer.divisor_bound}
  \leanok
  $d(n) = O_{\varepsilon}(n^{\varepsilon})$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} The prime-by-prime bound is
  $a + 1 \leq w(p) \cdot (p^a)^{\varepsilon}$. Once $p^{\varepsilon} \geq 2$ one may take
  $w(p) = 1$, since $a + 1 \leq 2^a \leq (p^{\varepsilon})^a$; below that threshold take
  $w(p) = C_{\varepsilon}$, the constant of \texttt{exists\_linear\_le\_pow} at
  $r = 2^{\varepsilon}$, which works uniformly in $p$ because $p \geq 2$. Only primes
  below the threshold contribute a factor and there are at most $P$ of them, so the
  constant is $C_{\varepsilon}^{P}$.

  The analytic input is that linear is dominated by exponential with an explicit
  constant, which is obtained by noting $(a+1)/r^a \to 0$ and taking the supremum of a
  convergent sequence.
\end{proof}

\begin{lemma}[Dirichlet swap]
  \label{lem:dirichlet-swap}
  \lean{DuffinSchaeffer.sum_sum_divisors}
  \leanok
  For any $f$, $\sum_{q \leq n} \sum_{d \mid q} f(d) = \sum_{d \leq n} \lfloor n/d \rfloor f(d)$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved.} The divisors of $q$ are exactly the $d \leq n$ dividing $q$ once
  $0 < q \leq n$ (\texttt{divisors\_eq\_filter}), so both sides count the same pairs;
  exchanging the order of summation and counting multiples with
  \texttt{Nat.Ioc\_filter\_dvd\_card\_eq\_div} gives the result.
\end{proof}

\begin{lemma}[Average order of $\varphi$]
  \label{lem:totient-average}
  \lean{DuffinSchaeffer.sum_totient_div_self}
  \leanok
  \uses{lem:dirichlet-swap}
  $\tfrac{n}{2} \leq \sum_{q \leq n} \varphi(q)/q \leq n$; in particular
  $\sum_{q \leq n} \varphi(q)/q \asymp n$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved}, with explicit constants $c = 1/2$ and $C = 1$.

  The upper bound is $\varphi(q) \leq q$ termwise. For the lower bound, sum
  $\sum_{d \mid q} \varphi(d) = q$ (Mathlib's \texttt{Nat.sum\_totient}) over $q \leq n$.
  The left side is $n(n+1)/2$ by Gauss; Lemma \ref{lem:dirichlet-swap} rewrites it as
  $\sum_{d \leq n} \lfloor n/d \rfloor \varphi(d)$, which is at most
  $n \sum_{d \leq n} \varphi(d)/d$. Dividing by $n$ gives
  $\sum_{q \leq n} \varphi(q)/q \geq (n+1)/2$.

  Worth recording because the obvious route is much worse: expanding
  $\varphi(q)/q = \sum_{d \mid q} \mu(d)/d$ leads to $\sum_{d} \mu(d)/d^2 = 1/\zeta(2)$
  and the Basel problem, none of which is needed. The constant $1/2$ is not sharp --- the
  truth is $6/\pi^2 \approx 0.61$ --- but no estimate downstream cares.
\end{proof}

\begin{lemma}[Pollington--Vaughan]

\end{remark}

\chapter{GCD graphs}
\label{chap:gcd}

The machinery of the paper, \S\S3--10. The overlap of $\Aq$ and $A_r$ is inflated by
the small primes dividing $qr/\gcd(q,r)^2$, so bounding the double sum means
controlling how often a set of denominators can be pairwise entangled in that sense.
Koukoulopoulos and Maynard encode the entanglement as a weighted graph, attach a
numerical \emph{quality} to it, and show that a graph which is bad for the overlap
estimate admits a modification strictly increasing its quality. Quality is bounded
above, so the iteration terminates, and the terminal graphs are exactly those for
which the estimate holds.

\begin{definition}[Bipartite GCD graph]
  \label{def:gcdgraph}
  \lean{DuffinSchaeffer.GCDGraph}
  \leanok
  \textbf{Transcribed} --- paper Definition 6.1. A septuple $(\mu, V, W, E, P, f, g)$
  where $\mu$ is a measure on $\N$ (extended to pairs by
  $\mu(N) = \sum_{(n_1,n_2)\in N}\mu(n_1)\mu(n_2)$), $V, W$ are finite sets of positive
  integers, $E \subseteq V \times W$, $P$ is a set of primes, and
  $f, g : P \to \Z_{\geq 0}$ satisfy, for every $p \in P$:
  \begin{enumerate}
    \item $p^{f(p)} \mid v$ for all $v \in V$, and $p^{g(p)} \mid w$ for all $w \in W$;
    \item if $(v,w) \in E$ then $p^{\min(f(p),g(p))} \,\|\, \gcd(v,w)$;
    \item if $f(p) \neq g(p)$ then $p^{f(p)} \,\|\, v$ for all $v \in V$ and
      $p^{g(p)} \,\|\, w$ for all $w \in W$.
  \end{enumerate}
\end{definition}

\begin{remark}
  An earlier draft of this blueprint carried a version of this definition written from
  memory. Every substantive feature of it was wrong: the graphs are \emph{bipartite}
  rather than having one vertex set with a symmetric edge relation; a \emph{measure} is
  part of the data, and the entire argument is about the weighted edge density $\mu(E)$;
  condition (1) is divisibility, not exact divisibility; condition (2) involves
  $\min(f,g)$, not $g$; condition (3) --- exactness, but \emph{only} when
  $f(p) \neq g(p)$ --- was missing altogether; and the guess imposed $g \leq f$, which the
  paper does not.

  This is what the \textbf{TRANSCRIBE} marks are for. Anything built on the earlier
  structure would have been unsound in a way invisible until someone opened the paper.
\end{remark}

  \label{def:gcdsubgraph}
  \lean{DuffinSchaeffer.GCDGraph.Nontrivial, DuffinSchaeffer.GCDGraph.IsSubgraph}
  \leanok
  \uses{def:gcdgraph}
  \textbf{Transcribed} --- Definitions 6.2 and 6.4. $G$ is non-trivial when $\mu(E) > 0$;
  and $G' \preccurlyeq G$ when $\mu' = \mu$, $V' \subseteq V$, $W' \subseteq W$,
  $E' \subseteq E$, $P' \supseteq P$ and $f', g'$ restrict to $f, g$ on $P$ --- note the
  set of primes \emph{grows}, since a subgraph carries more divisibility information.
\end{definition}

\begin{definition}[Prime-power subgraphs, and the quantities of a GCD graph]
  \label{def:quality}
  \lean{DuffinSchaeffer.GCDGraph.powPart, DuffinSchaeffer.GCDGraph.edgeDensity,
        DuffinSchaeffer.GCDGraph.neighborsV, DuffinSchaeffer.GCDGraph.R,

  \leanok

  \textbf{Transcribed} --- Definitions 6.5 and 6.6. Writing $V_{p^k}$ for the $v \in V$
  with $p^k \,\|\, v$, the edge density is $\delta = \mu(E)/(\mu(V)\mu(W))$, the
  neighbourhoods are the usual ones, $\mathcal{R}(G)$ is the set of primes outside $P$ dividing
  some $\gcd(v,w)$ along an edge, $\mathcal{R}^{\sharp}(G)$ those $p \in \mathcal{R}(G)$ for which a single

  \[ q(G) := \delta^{10}\mu(V)\mu(W)
     \prod_{p \in P}

\end{definition}

  so it is load-bearing rather than cosmetic.

  Note also that the quality lives in \S6, not \S7: this blueprint has now guessed its

\end{remark}

\begin{theorem}[Existence of a good GCD subgraph]
  \label{thm:gcd-iteration}
  \notready
  \uses{def:quality, def:gcdsubgraph, def:section5, prop:iter1, prop:iter2,
        prop:small-primes, lem:cosmetic, lem:high-degree}
  \textbf{Transcribed} --- Proposition 7.1. Let $G$ have trivial set of primes, edge
  density $\delta > 0$, every edge satisfying $L_t(v,w) \geq 10$, and
  $t \geq 10\delta^{-1/50}$ with $t > 10^{2000}$. Then $G$ has a GCD subgraph $G'$ of
  positive edge density with
  \begin{enumerate}
    \item $\mathcal{R}(G') = \emptyset$;
    \item $\mu(\Gamma_{G'}(v)) \geq (9\delta'/10)\mu(W')$ for all $v \in V'$, and
      symmetrically on the other side;
    \item either $q(G') \gg \delta t^{50} q(G)$, or $q(G') \gg q(G)$ and every edge still
      satisfies $L_t(v',w') \geq 4$ after the $P'$-part is stripped off.
  \end{enumerate}
  Sections 8--14 prove it; \S7 deduces Proposition 6.3 from it.
\end{theorem}

\begin{remark}
  In clause (3) the $\delta$ \emph{multiplies}. The published PDF renders that clause as
  \texttt{q(G')}$\gg\delta$\texttt{t50q(G)}, which reads naturally as $\gg_\delta$ --- an
  implied constant depending on $\delta$ --- and the arXiv source settles it as
  \verb|\gg \delta t^{50}|. The two are very different statements, and the weaker reading
  would have silently discarded the density gain that the entire compression argument
  exists to produce.
\end{remark}

\begin{theorem}[Duffin--Schaeffer]
  \label{thm:one}
  \notready
  \uses{prop:positive-suffices, prop:quasi-bc, thm:overlap-core, lem:seta-limsup}
  If $\sum \ps(q)\varphi(q)/q$ diverges then $\A(\ps)$ is measurable and
  $\Leb(\A(\ps)) = 1$.
\end{theorem}

\begin{proof}
  \uses{lem:truncation-gap}
  Put $\ps' = \min(\ps, 1/2)$, so $\A(\ps') \subseteq \A(\ps)$ by Lemma
  \ref{lem:seta-mono}. If $\sum \ps'(q)\varphi(q)/q$ still diverges, the argument runs on
  $\ps'$: Lemma \ref{lem:divergence} gives $\sum_q \Leb(A_q) = \infty$, Theorem
  \ref{thm:overlap-core} supplies the sets $S_n$ and the constant $C$, Proposition
  \ref{prop:quasi-bc} gives $\Leb(\A(\ps')) \geq C^{-1} > 0$ via Lemma
  \ref{lem:seta-limsup}, and Proposition \ref{prop:positive-suffices} upgrades that to
  $1$. Monotonicity finishes.

  If it does not diverge, Lemma \ref{lem:truncation-gap} is what is needed, and it is not
  proved here. Lean: \texttt{theorem\_1}.
\end{proof}

\begin{lemma}[The Duffin--Schaeffer conjecture for large values]
  \label{lem:large-values}
  \notready
  \textbf{External citation.} If $\ps(q) \in \{0\} \cup [1/2,\infty)$ for every $q$ and
  $\sum \ps(q)\varphi(q)/q$ diverges, then $\Leb(\A(\ps)) = 1$.

  This is Lemma 5.2 of the paper, and the paper does not prove it either: it is deduced
  from Theorem 2 of Pollington and Vaughan, \emph{The $k$-dimensional Duffin and Schaeffer
  conjecture}, Mathematika 37 (1990), 190--200. Formalizing it means formalizing that
  paper's Theorem 2.
\end{lemma}

\begin{lemma}[Reduction to $\ps \leq 1/2$]
  \label{lem:truncation-gap}
  \lean{DuffinSchaeffer.volume_setA_eq_one_of_psiLarge_diverges}
  \leanok
  \uses{lem:seta-mono, lem:large-values}
  Write $\ps = \ps_1 + \ps_2$ with $\ps_1 = \ps \cdot \mathbf{1}[\ps > 1/2]$ and
  $\ps_2 = \ps \cdot \mathbf{1}[\ps \leq 1/2]$. If $\sum \ps_1(q)\varphi(q)/q$ diverges
  then $\Leb(\A(\ps)) = 1$.
\end{lemma}

\begin{proof}
  \leanok
  \textbf{Proved}, taking Lemma \ref{lem:large-values} as an explicit hypothesis rather
  than smuggling it in --- the same pattern as Lemma \ref{lem:truncate} with the zero-one
  law. Given it, the argument is two lines: $\ps_1$ satisfies its hypothesis by
  construction, and $\ps_1 \leq \ps$ gives $\A(\ps_1) \subseteq \A(\ps)$.

  \emph{An earlier draft of this blueprint guessed this reduction and guessed wrong.} It
  used the truncation $\min(\ps, 1/2)$; the paper (\S5) splits $\ps$ additively by
  restriction instead, which is a different function --- where $\ps(q) = 5$ the truncation
  gives $1/2$ but $\ps_2$ gives $0$. That matters, because with the restriction split the
  two cases are genuinely disjoint and the large one is a known theorem
  (Lemma \ref{lem:large-values}) rather than something to be reconstructed. The earlier
  draft's worry --- that the divergence could hide in a regime nothing settles --- was
  real, and this is how the paper disposes of it.
\end{proof}

\begin{theorem}[Theorem 2(a), convergence]
  \label{thm:two-a}
  \lean{DuffinSchaeffer.theorem_2_a}
  \leanok
  \uses{def:psistar, def:seta}
  If $\sum_q \pstar(q) < \infty$ then $\K(\ps)$ is measurable and $\Leb(\K(\ps)) = 0$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:measurable-seta, lem:psisup-le-psistar}
  \textbf{Proved.} The first terminal node of the project to close. What follows is the
  argument as formalised.
  Measurability is done (Lemma \ref{lem:measurable-seta}). For the null statement:

  Given $\alpha \in \K(\ps)$ there are infinitely many pairs $(a,q)$, $a \leq q$, with
  $|\alpha - a/q| \leq \ps(q)/q$. Reduce $a/q$ to lowest terms $b/d$, $d \mid q$; then
  $|\alpha - b/d| \leq \ps(q)/q \leq \Psi(d)$. So $\alpha$ lies in
  \[ E_d := \bigcup_{\substack{0 \le b \le d \\ \gcd(b,d)=1}} \overline{B}(b/d, \Psi(d)), \qquad
     \Leb(E_d) \leq 2(\varphi(d)+1)\Psi(d) \leq 4\pstar(d), \]
  the last step by Lemma \ref{lem:psisup-le-psistar}. Since $\sum_d \pstar(d) < \infty$,
  the first Borel--Cantelli lemma (Mathlib:
  \texttt{MeasureTheory.measure\_limsup\_cofinite\_eq\_zero}) kills $\limsup_d E_d$.

  \textbf{The gap.} That argument needs infinitely many \emph{distinct} $d$, and the
  reduction to lowest terms does not supply them: infinitely many pairs can reduce to a
  single $b/d$. Take $\alpha = 0$, where every pair $(0,q)$ reduces to $0/1$. So the
  step ``infinitely many pairs, hence infinitely many $d$'' is false as stated.

  Two repairs are needed, and both are cheap once seen. First, discard the rationals:
  $\mathbb{Q}$ is countable, hence null, so it suffices to treat irrational $\alpha$.
  Second, note that $\sum_q \pstar(q) < \infty$ forces $\pstar(q) \to 0$, and
  $\pstar(q) \geq \Psi(q) \geq \ps(q)/q$, so $\ps(q)/q \to 0$. For irrational $\alpha$
  and fixed $b/d$ we have $|\alpha - b/d| > 0$, so only finitely many $q$ can satisfy
  $\ps(q)/q \geq |\alpha - b/d|$ --- that is, each $(b,d)$ absorbs only finitely many
  pairs. Infinitely many pairs therefore do give infinitely many $(b,d)$, and since
  $b \leq d$ bounds the numerators, infinitely many $d$.

  Worth stating plainly because the flawed version is the one that comes to mind first
  and it is not visibly wrong.

  Both repairs are in the Lean proof: \texttt{finite\_ratio\_ge} is the statement that
  $\{q : \delta \le \ps(q)/q\}$ is finite for $\delta > 0$, obtained from
  \texttt{ENNReal.finite\_const\_le\_of\_tsum\_ne\_top}, and the rationals are removed by
  \texttt{Set.Countable.measure\_zero}. The reduction to lowest terms is
  \texttt{redNum}/\texttt{redDen}, and the final step is Mathlib's
  \texttt{measure\_limsup\_cofinite\_eq\_zero} composed with
  \texttt{Filter.cofinite.limsup\_set\_eq}.
\end{proof}

\chapter{Out of scope: Hausdorff dimension}

\begin{definition}[Critical exponent]
  \label{def:critexp}
  \lean{DuffinSchaeffer.critExp}
  \leanok
  $s = \inf \{ \beta : \sum_q \varphi(q)(\ps(q)/q)^{\beta} < \infty \}$.
\end{definition}

\begin{corollary}[Corollary 3 of the paper]
  \label{cor:three}
  \notready
  \uses{thm:one, def:critexp}
  For $\ps$ taking values in $[0,1/2]$, $\dimH \A(\ps) = \min(s,1)$.
  Lean: \texttt{corollary\_3}.
\end{corollary}

This does not follow from Theorem \ref{thm:one} by soft arguments. The passage from
full Lebesgue measure to full Hausdorff dimension runs through the mass transference
principle of Beresnevich and Velani (\emph{Annals} 164 (2006), 971--992), which
Mathlib does not have --- searching for \texttt{massTransference} or
\texttt{beresnevich} returns nothing --- and which brings its own prerequisite stack:
Hausdorff content, a Cantor-set construction, the ubiquity framework.

That is a separate project of comparable size which happens to share a statement file.
Folding it into the critical path would roughly double the work and add nothing to the
conjecture. Mathlib does have $\dimH$ and $\mu H[d]$ with a workable API, so the
landing site exists whenever someone wants it.
